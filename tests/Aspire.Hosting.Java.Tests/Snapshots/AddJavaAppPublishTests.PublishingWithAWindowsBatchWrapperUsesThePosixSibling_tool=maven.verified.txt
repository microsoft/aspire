FROM --platform=$BUILDPLATFORM docker.io/library/eclipse-temurin:21-jdk AS build
WORKDIR /app
COPY scripts/custom-mvnw ./scripts/custom-mvnw
COPY scripts/.mvn ./scripts/.mvn
RUN sed -i 's/\r$//' ./scripts/custom-mvnw && sh ./scripts/custom-mvnw -B -ntp --version
COPY . .
RUN --mount=type=cache,id=aspire-java-maven-api,target=/root/.m2/repository,sharing=locked \
    mkdir -p /build && sed -i 's/\r$//' ./scripts/custom-mvnw && sh ./scripts/custom-mvnw -B -ntp -DskipTests package && jars=$(ls target/*.jar 2>/dev/null | grep -Ev '(-plain|-sources|-javadoc)\.jar$' || true) && count=$(printf '%s\n' "$jars" | grep -c . || true) && if [ "$count" != "1" ]; then echo "Aspire: expected exactly one application JAR from the build of 'api' matching target/*.jar, found $count:" >&2; echo "$jars" >&2; echo "Use WithJarArtifact(\"<relative path>\") to select one." >&2; exit 1; fi && cp "$jars" /build/app.jar

FROM docker.io/library/eclipse-temurin:21-jre
WORKDIR /app
COPY --from=build --chown=999:999 /build/app.jar /app/app.jar
USER 999:999
ENTRYPOINT ["java","-jar","/app/app.jar"]
