FROM node:22-slim
WORKDIR /app
ARG NPM_CONFIG_REGISTRY=https://pkgs.dev.azure.com/dnceng/public/_packaging/dotnet-public-npm/npm/registry/
RUN npm install --global pnpm@10.30.1
COPY package.json ./
RUN --mount=type=cache,target=/pnpm/store pnpm install --prefer-frozen-lockfile
COPY . .
RUN pnpm run mybuild
