FROM denoland/deno:2 AS build
WORKDIR /app
COPY deno.json ./
RUN --mount=type=cache,target=/deno-dir deno install
COPY . .
RUN deno task build

FROM denoland/deno:2 AS runtime
WORKDIR /app
COPY --from=build /app /app
ENV NODE_ENV=production
ENTRYPOINT ["sh","-c","exec deno task start -- --port $PORT"]
