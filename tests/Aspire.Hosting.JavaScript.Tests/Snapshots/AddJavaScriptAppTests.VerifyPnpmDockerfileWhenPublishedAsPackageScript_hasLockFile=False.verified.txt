FROM node:22-slim AS build
WORKDIR /app
ARG NPM_REGISTRY=https://pkgs.dev.azure.com/dnceng/public/_packaging/dotnet-public-npm/npm/registry/
RUN npm install --global --registry "$NPM_REGISTRY" pnpm@10.30.1
COPY package.json ./
RUN --mount=type=cache,target=/pnpm/store pnpm install --prefer-frozen-lockfile
COPY . .
RUN pnpm run mybuild

FROM node:22-slim AS prod-deps
WORKDIR /app
ARG NPM_REGISTRY=https://pkgs.dev.azure.com/dnceng/public/_packaging/dotnet-public-npm/npm/registry/
RUN npm install --global --registry "$NPM_REGISTRY" pnpm@10.30.1
COPY package.json ./
RUN --mount=type=cache,target=/pnpm/store pnpm install --prefer-frozen-lockfile --prod

FROM node:22-alpine AS runtime
WORKDIR /app
COPY --from=build /app /app
COPY --from=prod-deps /app/node_modules ./node_modules
ARG NPM_REGISTRY=https://pkgs.dev.azure.com/dnceng/public/_packaging/dotnet-public-npm/npm/registry/
RUN npm install --global --registry "$NPM_REGISTRY" pnpm@10.30.1
ENV NODE_ENV=production
ENTRYPOINT ["sh","-c","exec pnpm run start"]
